function p.BuyCore()
    local r=h() if not r then return false,"no save data yet" end
    if not D or type(D.CoreNodes)~="table" or type(D.CoreUpgradeById)~="table" then return false,"core config unavailable" end
    local bought,refused,poor,lockBusy=0,0,0,false
    for _=1,60 do
        local pick
        for _,nd in ipairs(D.CoreNodes) do
            local up=D.CoreUpgradeById[nd.Family]
            if up and up.Event then
                local okM,maxed=pcall(D.IsCoreNodeMaxed,r,nd) maxed=(not okM) or maxed -- [แก้ 12] อ่านไม่ได้ = ถือว่าเต็ม
                local okO,open=pcall(D.IsCoreNodeOpen,r,nd) open=okO and open -- [แก้ 12] อ่านไม่ได้ = ไม่เปิด
                if open and not maxed then
                    local okL,lv=pcall(D.GetCoreLevel,r,nd.Family) lv=(okL and tonumber(lv)) or 0
                    local _,cost=pcall(up.Cost,lv)
                    if lv<(tonumber(up.Max) or math.huge) then
                        cost=tonumber(cost) or math.huge
                        if cost<=Y() then pick={fam=nd.Family,ev=up.Event,lvl=lv,cost=cost} break else poor=poor+1 end
                    end
                end
            end
        end
        if not pick then break end
        -- [แก้ 6] จองล็อกเงินระหว่างยิง + เช็คผล
        local got,res=p.__withLock("core",function()
            if Y()<pick.cost then return "poor" end
            H(pick.ev) task.wait(.45)
            r=h() or r
            local _,nl=pcall(D.GetCoreLevel,r,pick.fam)
            return ((tonumber(nl) or 0)>pick.lvl) and "ok" or "refused"
        end)
        if not got then lockBusy=true break end
        if res=="ok" then bought=bought+1 elseif res=="poor" then poor=poor+1 break else refused=refused+1 break end
    end
    if bought>0 then return true,("bought %d core upgrade(s)"):format(bought) end
    if lockBusy then return false,"money lock busy" end
    if refused>0 then return false,"server refused a core purchase" end
    if poor>0 then return false,"cannot afford any core upgrades yet" end
    return false,"no core upgrades left"
end

-- [แก้ 6] ล็อกเงินกลาง: ระบบที่ทำให้เงินเปลี่ยน (อัพตัว / Core / สกิล / อีเตอร์ / ปลดโซน / Mutation / ขาย) ยิงทีละระบบตามคิว
--   รอคิวได้สูงสุด 5 วิ (เกิน = ข้ามครั้งนี้), จองจนเช็คผลครั้งนั้นเสร็จ, ปล่อยทุกทางออกแม้ error
p.__spendLock={next=0,serving=1,gone={},holder=nil}
p.__withLock=function(tag,fn)
    local L=p.__spendLock
    L.next=L.next+1
    local my=L.next
    local t0=os.clock()
    local function advance()
        L.serving=L.serving+1
        while L.gone[L.serving] do L.gone[L.serving]=nil L.serving=L.serving+1 end
    end
    while L.serving~=my do
        if L.holder and os.clock()-L.holder.at>15 then L.holder=nil advance() -- คนถือค้างเกิน 15 วิ (ไม่ควรเกิด) → ข้ามคิวนั้น
        elseif os.clock()-t0>=5 then L.gone[my]=true return false end
        if L.serving~=my then task.wait(0.05) end
    end
    L.holder={tag=tag,at=os.clock()}
    local res=table.pack(pcall(fn))
    L.holder=nil
    advance()
    if not res[1] then error(res[2],0) end
    return true,table.unpack(res,2,res.n)
end
-- [แก้ v16] อัพตัวเฉพาะ: ทุกตัวบนแท่นที่ระดับ Mythical ขึ้นไป (ตามลำดับระดับของเกม) และเป็นตัวของโซนลึกสุดที่ปลดแล้ว
--   ต่ำกว่า Mythical / ตัวโซนเก่า / อ่านระดับ-ลำดับระดับ-โซนไม่ได้ = ห้ามอัพ
p.__crewRarity=function(data)
    if type(data)~="table" then return nil end
    local idx=D and D.RarityIndex
    local function ok(v) return type(v)=="string" and type(idx)=="table" and idx[v]~=nil end
    local keys={"Rarity","rarity","RarityName","Tier"}
    local s0,rv=pcall(function() return data.Rarity end)
    if s0 and type(rv)=="table" then
        for _,k in ipairs({"Rarity","Name","Id","Key"}) do local v=rawget(rv,k) if ok(v) then return v end end
    end
    for _,k in ipairs(keys) do
        local s1,v=pcall(function() return data[k] end)
        if s1 and ok(v) then return v end
    end
    for _,sub in pairs(data) do
        if type(sub)=="table" then
            for _,k in ipairs(keys) do local v=rawget(sub,k) if ok(v) then return v end end
        end
    end
    return nil
end
p.__crewZone=function()
    local r0=h()
    if not r0 or not D then return nil,nil end
    local deepest
    for _,z in ipairs(D.ZoneKeys or {}) do if (r0.UnlockedZones or {})[z] then deepest=z end end
    if not deepest then return nil,nil end
    local zd=type(D.Zones)=="table" and D.Zones[deepest]
    local pool=type(zd)=="table" and zd.CharacterPool
    if type(pool)~="table" then return deepest,nil end
    local set,list={},{}
    for _,e in pairs(pool) do
        local nm=(type(e)=="table" and e.Name) or (type(e)=="string" and e) or nil
        if type(nm)=="string" and not set[nm] then set[nm]=true list[#list+1]=nm end
    end
    if #list==0 then return deepest,nil end
    return deepest,set,list
end
p.__zoneTargets=function(zk,list)
    p.__tgtCache=p.__tgtCache or {}
    if p.__tgtCache[zk] then return p.__tgtCache[zk] end
    local out={}
    for _,nm in ipairs(list or {}) do
        local rr=p.__crewRarity(N({CharName=nm}))
        local rk,mr=rr and p.__rarityRank(rr),p.__minRank()
        if rk and mr and rk>=mr then out[#out+1]=nm.." ("..rr..")" end
    end
    table.sort(out)
    p.__tgtCache[zk]=out
    return out
end
-- ลำดับระดับจากเกม (เลขยิ่งมากยิ่งหายาก): RarityIndex ก่อน ไม่ได้ค่อยใช้ RarityOrder / อ่านไม่ได้ = nil
p.__rarityRank=function(name)
    if type(name)~="string" or not D then return nil end
    local idx=D.RarityIndex
    local v=type(idx)=="table" and tonumber(idx[name]) or nil
    if v then return v end
    local ord=D.RarityOrder
    if type(ord)=="table" then for i,n0 in ipairs(ord) do if n0==name then return i end end end
    return nil
end
-- ระดับขั้นต่ำที่อัพ (CONFIG.crew_min_rarity) — ลำดับอ่านไม่ได้หรือกลับด้าน (Common สูงกว่า) = nil → ไม่อัพเลย
p.__minRank=function()
    local nm=tostring(b.Flags.crew_min_rarity or "Mythical")
    local mr=p.__rarityRank(nm)
    local cr=p.__rarityRank("Common")
    if not mr or (cr and cr>=mr) then return nil,nm end
    return mr,nm
end
p.__crewAllowed=function(list)
    local r0=h()
    local holders=(r0 and r0.Holders) or {}
    local zk,pool,plist=p.__crewZone()
    if not pool then
        return {},(zk and ("cannot read "..tostring(zk).." character pool - nothing upgraded") or "cannot read zones - nothing upgraded"),{},0,zk,nil
    end
    local minR,minName=p.__minRank()
    if not minR then return {},"cannot read rarity order - nothing upgraded",{},0,zk,plist end
    local keys,picked,unknown={},{},0
    for _,e in ipairs(list) do
        local hd=holders[e.key]
        e.rarity=p.__crewRarity(N(hd))
        e.inZone=(type(hd)=="table" and pool[hd.CharName]) and true or false
        local rk=e.rarity and p.__rarityRank(e.rarity)
        if not rk then unknown=unknown+1
        elseif e.inZone and rk>=minR then keys[#keys+1]=e.key picked[#picked+1]=e.key.." "..e.rarity end
    end
    local why
    if #keys==0 then
        if #list>0 and unknown==#list then why="cannot read crew rarity - nothing upgraded"
        else why="no "..minName.."+ from "..tostring(zk).." in your crew" end
    end
    return keys,why,picked,unknown,zk,plist
end
-- [สมองกลาง] ใช้ตัดสินใจร่วมกันทั้งหน้าจอ HUD และระบบอัพตัว
--   buy   = มีสกิลที่พลังถึงและเงินพอ → ซื้อสกิล (ไม่อัพตัว)
--   save  = มีสกิลที่พลังถึงแต่เงินไม่พอ → เก็บเงินซื้อสกิล (ไม่อัพตัว)
--   power = สกิลที่เหลือติดพลังทีมทั้งหมด → ใช้เงินเพิ่มพลัง (อัพตัวทุกสแตท + อีเตอร์ + Core)
--   zone  = วิจัยครบ ยังไม่ปลดโซนถัดไป → เก็บเงินปลดโซน (ไม่อัพตัว ถ้า crew_wait_zone_unlock)
--   free  = ไม่มีอะไรให้ซื้อ → ใช้เงินได้อิสระ
-- [แก้ v16-v18] โหนดลำดับแรก = โหนดที่ชื่อ/Family/Id หรือคำอธิบายของเกม (MineConfig.SkillById[Family]) มีคำใน CONFIG.research_first
p.__isDmgNode=function(zk,nd)
    local words=b.Flags.research_first
    if type(words)~="table" or #words==0 or type(nd)~="table" then return false end
    local info
    pcall(function() info=type(D.SkillById)=="table" and D.SkillById[nd.Family] or nil end)
    local src={nd,nd,nd,nd,info,info}
    for i,k in ipairs({"Family","Name","Id","DisplayName","DisplayName","Desc"}) do
        local t0=src[i]
        local okV,v0=pcall(function() return type(t0)=="table" and t0[k] or nil end)
        if okV and (type(v0)=="string" or type(v0)=="number") then
            local lv0=tostring(v0):lower()
            for _,w in ipairs(words) do
                if type(w)=="string" and w~="" and lv0:find(w:lower(),1,true) then return true end
            end
        end
    end
    return false
end
-- ไล่โหนดที่ซื้อต่อได้ทุกโซนที่ปลด (เปิด ไม่เต็ม เลเวล<สูงสุด อ่านราคาได้) + พลังที่ต้องใช้ของเลเวลถัดไป
p.__skillScan=function(r0)
    local out,power={},nil
    local function team()
        if power==nil then local okT,tp=pcall(D.GetTeamPower,r0,m and m.GetItemData) power=(okT and tonumber(tp)) or 0 end
        return power
    end
    for _,zk in ipairs(D.ZoneKeys or {}) do
        if (r0.UnlockedZones or {})[zk] then
            local sk=(r0.Skills or {})[zk] or {}
            local okN,nodes=pcall(D.GetSkillNodes,zk)
            for _,nd in ipairs((okN and type(nodes)=="table" and nodes) or {}) do
                local okM,mx=pcall(D.IsSkillNodeMaxed,sk,nd) mx=(not okM) or mx -- [แก้ 12] อ่านไม่ได้ = ถือว่าเต็ม
                local okO,op=pcall(D.IsSkillNodeOpen,sk,nd) op=okO and op -- [แก้ 12] อ่านไม่ได้ = ไม่เปิด
                if op and not mx then
                    local lv=tonumber(sk[nd.Family]) or 0
                    local okX,maxLv=pcall(D.GetSkillMax,zk,nd.Family)
                    local okC,cost=pcall(D.GetSkillCost,zk,lv,nd.Family)
                    cost=okC and tonumber(cost) or nil
                    if okX and cost and lv<(tonumber(maxLv) or 0) then
                        local req
                        if type(D.GetSkillPowerReq)=="function" then
                            local okR,rq=pcall(D.GetSkillPowerReq,zk,nd.Family,lv+1)
                            req=okR and tonumber(rq) or nil
                        end
                        out[#out+1]={zone=zk,node=nd,name=tostring(nd.Name or nd.Family or nd.Id),fam=nd.Family,lv=lv,cost=cost,req=req,
                            dmg=p.__isDmgNode(zk,nd),key=zk..":"..tostring(nd.Id or nd.Family)..":"..lv}
                    end
                end
            end
        end
    end
    return out,team
end
-- [แก้ v19] อัพตัวเพิ่มพลังได้ไหม: เปิดอัพตัว + มีตัวผ่านกฎ + อย่างน้อยหนึ่งสแตทยังไม่ตัน (ไม่ดูเงิน — เงินไม่พอแค่รอ)
p.__crewCanGrow=function()
    if not b.Flags.auto_crew then return false,"auto_crew off" end
    if not D then return false,"game modules not ready" end
    local okA,list=pcall(A)
    if not okA or type(list)~="table" or #list==0 then return false,"no crew slots yet" end
    local keys,why=p.__crewAllowed(list)
    if #keys==0 then return false,why end
    -- เกมไม่ให้อัพตัวรอบล่าสุด (ภายใน 60 วิ ยังไม่มีครั้งที่สำเร็จหลังจากนั้น) → ถือว่าอัพไม่ได้ชั่วคราว กันสกิลค้างรอ
    if p.__crewRefusedAt and os.clock()-p.__crewRefusedAt<60 and not (p.__crewOkAt and p.__crewOkAt>p.__crewRefusedAt) then
        return false,"server refused the upgrade"
    end
    local r0=h()
    if not r0 then return false,"waiting for game data" end
    local stats=b.Flags.crew_stats
    if type(stats)~="table" or #stats==0 then stats={"Damage"} end
    for _,k in ipairs(keys) do
        for _,st in ipairs(stats) do
            local okR,v0=pcall(R,r0,k,st)
            if okR and v0 then return true end
        end
    end
    return false,"crew capped"
end
-- เลือกโหนด: โหนดลำดับแรกที่พลังถึง (ถูกสุด) ก่อน → ไม่มีค่อยเอาโหนดอื่นที่พลังถึง (ถูกสุด)
--   ไม่มีที่พลังถึงเลย → โหนดติดพลัง (ลำดับแรกก่อน แล้วค่อยที่ต้องใช้พลังน้อยสุด) = โหมด power
p.__skillChoose=function(list,team)
    local dOK,oOK,dPow,oPow
    for _,c in ipairs(list) do
        if not c.req or c.req<=team() then
            if c.dmg then if not dOK or c.cost<dOK.cost then dOK=c end
            elseif not oOK or c.cost<oOK.cost then oOK=c end
        else
            if c.dmg then if not dPow or c.req<dPow.req then dPow=c end
            elseif not oPow or c.req<oPow.req then oPow=c end
        end
    end
    local okPick,first=dOK or oOK,nil
    -- [แก้ v19] โหนดลำดับแรก (Density) ติดพลังทีม และไม่มีโหนดลำดับแรกที่ซื้อได้ → ไปเพิ่มพลังก่อน (โหมด power, ไม่ซื้อโหนดอื่น)
    --   ถ้าอัพตัวเพิ่มพลังไม่ได้ (ไม่มีตัวผ่านกฎ / ตันเลเวล / ปิดอัพตัว) → กลับไปซื้ออันถูกสุดตามเดิม กันค้าง
    if not dOK and dPow then
        local canGrow,why=p.__crewCanGrow()
        if canGrow then okPick=nil else first={node=dPow,why=why} end
    end
    return okPick,dPow or oPow,first
end
p.__brain=function()
    local r0=h()
    if not r0 or not D or type(D.GetSkillNodes)~="function" then return {mode="nodata"} end
    local money=Y()
    local list,team=p.__skillScan(r0)
    local bestOK,bestPow,first=p.__skillChoose(list,team)
    local power=(bestOK and bestOK.req) and team() or nil
    if bestOK then return {mode=(money>=bestOK.cost) and "buy" or "save",target=bestOK,money=money,power=power,first=first} end
    if bestPow then return {mode="power",target=bestPow,money=money,power=team()} end
    if type(D.ZoneResearchGateFor)=="function" and type(D.IsZoneResearchComplete)=="function" then
        for _,zk in ipairs(D.ZoneKeys or {}) do
            if not (r0.UnlockedZones or {})[zk] then
                local okG,gate=pcall(D.ZoneResearchGateFor,zk)
                local okC,done=pcall(D.IsZoneResearchComplete,r0.Skills,gate)
                if okG and okC and done then
                    local zc=nil
                    if type(p.__zoneCost)=="function" then local okZ,v0=pcall(p.__zoneCost,zk) zc=okZ and v0 or nil end
                    return {mode="zone",target={zone=zk,name="ปลดโซน "..zk,cost=zc,key="zone:"..zk},money=money}
                end
                break
            end
        end
    end
    return {mode="free",money=money}
end
-- [แก้ 1] แคชผลสมองกลาง 0.5 วิ (ระบบอัพตัวเช็คซ้ำก่อนยิงทุกครั้ง ไม่ให้คำนวณหนัก)
p.__brainCompute=p.__brain
p.__brain=function()
    local now=os.clock()
    if p.__brainCache and now>=p.__brainAt and now-p.__brainAt<0.5 then return p.__brainCache end
    local r0=p.__brainCompute()
    p.__brainCache,p.__brainAt=r0,now
    return r0
end
-- อัพตัวได้เฉพาะโหมด power / free (และ zone ถ้าปิด crew_wait_zone_unlock)
p.__treeCanProgress=function()
    local br=p.__brain()
    if br.mode=="nodata" then return true,"waiting for game data" end
    if br.mode=="buy" or br.mode=="save" then return true,"saving money for skill tree ("..tostring(br.target.zone).." "..tostring(br.target.name)..")" end
    if br.mode=="zone" and b.Flags.crew_wait_zone_unlock~=false then return true,"saving money to unlock "..tostring(br.target.zone) end
    return false,nil
end

function p.UpgradeCrew()
    -- [แก้ 8] เก็บผลล่าสุดไว้ให้ HUD: สำเร็จไหม / เหตุผล / เวลา / พักเพราะสกิลทรีหรือไม่
    local function fin(ok,why,busy)
        p.__crewLast={ok=ok and true or false,why=why,at=os.clock(),busy=busy and true or false}
        return ok,why
    end
    if not D or not m then return fin(false,"game modules not ready") end
    do local busy,why0=p.__treeCanProgress() if busy then p.__crewHold=why0 return fin(false,why0,true) end p.__crewHold=nil end
    local r=A()
    if #r==0 then return fin(false,"no crew slots yet") end
    local U=b.Flags.crew_stats
    if type(U)~="table" or #U==0 then U={"Damage"} end
    local B,why=p.__crewAllowed(r)
    if #B==0 then return fin(false,why) end
    local function Q(rr,money,stat)
        local best,score
        for _,key in ipairs(B) do
            local V=R(rr,key,stat)
            if V and V.cost<=money then
                local sc
                if stat=="Damage" then
                    local gain=f(rr,key,(rr.Holders or {})[key],V.level)
                    sc=gain and (gain/V.cost) or (1/V.cost)
                else
                    sc=1/V.cost
                end
                if not score or sc>score then best,score=V,sc end
            end
        end
        return best
    end
    local bought,spent,per=0,0,{}
    local fails,t0,rot=0,os.clock(),1
    local allowAt,stopWhy,stopBusy=os.clock(),nil,false
    while bought<300 and (os.clock()-t0)<30 do
        -- [แก้ 1] ทุก 2 วิ คำนวณรายชื่อตัวที่อนุญาตใหม่ (ช่องที่ไม่ผ่านกฎแล้ว = เลิกอัพ)
        if os.clock()-allowAt>=2 then
            allowAt=os.clock()
            local nb,w2=p.__crewAllowed(A())
            B=nb
            if #B==0 then stopWhy=w2 break end
        end
        local rr=h()
        if not rr then break end
        local money,pick=Y(),nil
        for _=1,#U do
            local st=U[rot]
            rot=(rot%#U)+1
            pick=Q(rr,money,st)
            if pick then break end
        end
        if not pick then break end
        -- [แก้ 1] ก่อนยิงทุกครั้ง: สกิลทรีกลับมาไปต่อได้ / ต้องเก็บเงิน → หยุดอัพตัวทันที
        local busy1,why1=p.__treeCanProgress()
        if busy1 then p.__crewHold=why1 stopWhy,stopBusy=why1,true break end
        -- [แก้ 6] จองล็อกเงินระหว่างยิง + เช็คผล
        local got,sent,before,after=p.__withLock("crew",function()
            local bf=Y()
            if not H("MineUpgradeCharacter",pick.key,pick.stat) then return false,bf,bf end
            task.wait(.3)
            return true,bf,Y()
        end)
        if not got then stopWhy="money lock busy" break end
        if not sent then break end
        if after<before then
            bought=bought+1
            spent=spent+(before-after)
            fails=0
            per[pick.stat]=(per[pick.stat] or 0)+1
            p.__crewOkAt=os.clock()
        else
            fails=fails+1
            if fails>=3 then break end
        end
    end
    if bought==0 then
        if stopWhy then return fin(false,stopWhy,stopBusy) end
        if fails>0 then p.__crewRefusedAt=os.clock() return fin(false,"server refused the upgrade") end -- [แก้ v19] จำเวลาไว้ให้ระบบวิจัยรู้
        return fin(false,("nothing affordable - %s in the bank"):format(a(Y())))
    end
    local K={}
    for _,st in ipairs(v) do if per[st] then K[#K+1]=("%d %s"):format(per[st],st) end end
    return fin(true,("bought %s for %s - %s left"):format(table.concat(K," + "),a(spent),a(Y())))
end

function p.AdvanceZone()
    local r=h() if not r or not D then return false,"no save data yet" end
    local keys=D.ZoneKeys or {}
    for idx=2,#keys do
        local zk=keys[idx]
        if not (r.UnlockedZones or {})[zk] then
            local okG,gate=pcall(D.ZoneResearchGateFor,zk)
            if not okG or not gate then break end
            local okC,done=pcall(D.IsZoneResearchComplete,r.Skills,gate)
            if okC and done then
                -- [แก้ 4] ยิงปลดโซนเฉพาะตอนอ่านราคาได้และเงินพอ / อ่านราคาไม่ได้ (หรือเพิ่งโดนปฏิเสธ) = ห่างกันอย่างน้อย 45 วิ
                local cost
                if type(p.__zoneCost)=="function" then local okZ,v0=pcall(p.__zoneCost,zk) cost=okZ and tonumber(v0) or nil end
                if cost and Y()<cost then return false,"saving money to unlock "..zk end
                local now=os.clock()
                if (not cost or p.__zoneRefused) and now-(p.__zoneTryAt or -1e9)<45 then return false,zk.." unlock: next try later" end
                p.__zoneTryAt=now
                -- [แก้ 6] จองล็อกเงินระหว่างยิง + เช็คผล
                local got,ok=p.__withLock("zone",function()
                    if cost and Y()<cost then return nil end
                    H("MineSelectZone",zk) task.wait(1.5)
                    r=h() or r
                    return ((r.UnlockedZones or {})[zk]) and true or false
                end)
                if not got then return false,"money lock busy" end
                if ok==nil then return false,"saving money to unlock "..zk end
                if ok then p.__zoneRefused=false return true,"unlocked "..zk end
                p.__zoneRefused=true
                return false,zk.." refused by the server"
            end
            local okP,a1,a2=pcall(D.GetZoneResearchProgress,r.Skills,gate)
            if okP then return false,("%s research %s/%s \226\128\148 max its skills first"):format(tostring(gate),tostring(a1),tostring(a2)) end
            return false,"waiting on "..tostring(gate).." research"
        end
    end
    local deepest
    for _,zk in ipairs(keys) do if (r.UnlockedZones or {})[zk] then deepest=zk end end
    if not deepest then return false,"no zones unlocked" end
    if tostring(r.CurrentZone)==tostring(deepest:match("%d+")) then return false,"already in the deepest zone" end
    -- [แก้ 4] ย้ายไปโซนลึกสุด ไม่ถี่กว่า 45 วิครั้ง (เท่ารอบเดิม)
    if os.clock()-(p.__zoneTravelAt or -1e9)<45 then return false,"travel: next try later" end
    p.__zoneTravelAt=os.clock()
    local got,sent=p.__withLock("zone",function() return H("MineSelectZone",deepest) end)
    if not got then return false,"money lock busy" end
    return sent,sent and ("travelled to "..deepest) or "travel failed"
end

-- ===== สุ่ม Mutation ให้ตัวที่อัพ (ตอนติดพลังทีมเท่านั้น, ตัวละครั้งเดียว) =====
-- [แก้ 5] อ่าน mutation แบบรู้ผล: "none" = เป็นค่าเริ่มต้นแน่ๆ (สุ่มได้) / "has" = มีแล้ว / "unknown" = อ่านไม่ได้ (ห้ามสุ่ม)
p.__variantState = function(hd)
    if type(hd) ~= "table" or not D or type(D.GetHolderVariant) ~= "function" then return "unknown" end
    local dv = D.DefaultVariant
    if dv == nil then return "unknown" end
    local ok, v0 = pcall(D.GetHolderVariant, hd)
    if not ok or type(v0) ~= "string" then return "unknown" end
    if v0 == dv then return "none" end
    return "has", v0
end
p.__mutRolled = {}
function p.CrewMutation()
    local br = p.__brain()
    if br.mode ~= "power" then return false, "not power-blocked" end
    if not D or type(D.GetMutationRollCost) ~= "function" then return false, "mutation config unavailable" end
    local keys = p.__crewAllowed(A())
    local rolled, lockBusy = 0, false
    for _, key in ipairs(keys) do
        if b.Unloaded then break end
        local r = h()
        local hd = r and (r.Holders or {})[key]
        if type(hd) == "table" and hd.CharName then
            local tag = key .. ":" .. tostring(hd.CharName)
            if not p.__mutRolled[tag] and p.__variantState(hd) == "none" then
                local okC, cost = pcall(D.GetMutationRollCost, N(hd), r)
                cost = okC and tonumber(cost) or nil
                if cost and cost <= Y() then
                    -- [แก้ 6] ถือล็อกเงินตั้งแต่ก่อนยิงจนเช็คผลเสร็จ → เงินที่ลดระหว่างนั้นคือค่าสุ่มจริง
                    local got, res = p.__withLock("mutation", function()
                        local r1 = h()
                        local cur = r1 and (r1.Holders or {})[key]
                        if type(cur) ~= "table" or tostring(cur.CharName) ~= tostring(hd.CharName) or p.__variantState(cur) ~= "none" then return "skip" end
                        local before = Y()
                        if before < cost then return "skip" end
                        if not H("MineRollMutation", key) then return "fail" end
                        task.wait(1.4)
                        local r2 = h()
                        local after = r2 and (r2.Holders or {})[key]
                        if p.__variantState(after) == "has" or Y() < before then return "rolled" end
                        return "fail"
                    end)
                    if not got then lockBusy = true break end
                    if res == "rolled" then p.__mutRolled[tag] = true rolled = rolled + 1 end
                end
            end
        end
    end
    if rolled > 0 then return true, ("rolled mutation on %d crew"):format(rolled) end
    if lockBusy then return false, "money lock busy" end
    return false, "nothing to roll"
end

-- วินาทีที่บัฟของกลุ่มนี้เหลือ (0 = ไม่มี/หมดแล้ว)
p.__potLeft=function(r0,fam)
    local bo=type(r0)=="table" and type(r0.Boosts)=="table" and r0.Boosts[fam]
    local u=type(bo)=="table" and tonumber(bo.Until) or nil
    if not u then return 0 end
    return math.max(0,u-p.__potNow())
end
function p.UsePotions()
    local r0=h()
    if not r0 then return false,"no save data yet" end
    local fams=b.Flags.potion_families
    if type(fams)~="table" or #fams==0 then return false,"no potion families picked" end
    local margin=tonumber(b.Flags.potion_margin) or 5
    p.__potBan=p.__potBan or {}
    local used,notes=0,{}
    local names={}
    for _,fam in ipairs(fams) do
        local owned=type(r0.Potions)=="table" and r0.Potions or {}
        if p.__potLeft(r0,fam)<=margin then
            local best,bestPow
            for nm,cnt in pairs(owned) do
                local d=(tonumber(cnt) or 0)>0 and p.__potDef(nm) or nil
                local ban=p.__potBan[nm]
                if type(d)=="table" and d.Family==fam and d.Kind=="Timed" and not (ban and os.clock()<ban.untilAt) then
                    local pw=tonumber(d.Mult) or tonumber(d.Add) or 0
                    if not best or pw>bestPow or (pw==bestPow and tostring(nm)<tostring(best)) then best,bestPow=nm,pw end
                end
            end
            if best then
                local c0,u0=tonumber(owned[best]) or 0,p.__potLeft(r0,fam)
                if H("MineUsePotion",best,1) then
                    -- รอผลจากเซิร์ฟสูงสุด ~3 วิ (เน็ตแลคก็ไม่นับเป็นเกมไม่รับ)
                    local r1,okUse=r0,false
                    for _=1,10 do
                        task.wait(0.3)
                        r1=h() or r1
                        local c1=tonumber((type(r1.Potions)=="table" and r1.Potions or {})[best]) or 0
                        if c1<c0 or p.__potLeft(r1,fam)>u0+1 then okUse=true break end
                    end
                    if okUse then
                        used=used+1 names[#names+1]=tostring(best)
                        p.__potBan[best]=nil
                    else
                        local ban=p.__potBan[best] or {fails=0,untilAt=0}
                        ban.fails=ban.fails+1
                        if ban.fails>=2 then ban.fails,ban.untilAt=0,os.clock()+300 end
                        p.__potBan[best]=ban
                        notes[#notes+1]="server refused "..tostring(best)
                    end
                    r0=r1
                else
                    notes[#notes+1]="send failed"
                end
            end
        end
    end
    if used>0 then return true,"used "..table.concat(names,", ") end
    if #notes>0 then return false,table.concat(notes,"; ") end
    return false,"buffs still running or no potions left"
end

    return on,#fams,banned
end

local function serverList()
    local list, cursor = {}, ""
    for _ = 1, 4 do
        local url = ("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100"):format(game.PlaceId)
        if cursor ~= "" then url = url .. "&cursor=" .. cursor end
        local ok, body = pcall(function() return game:HttpGet(url) end)
        if not ok then break end
        local ok2, data = pcall(function() return V:JSONDecode(body) end)
        if not ok2 or type(data) ~= "table" or not data.data then break end
        for _, s0 in ipairs(data.data) do
            if s0.id ~= game.JobId and s0.playing and s0.maxPlayers and s0.playing < s0.maxPlayers then list[#list + 1] = s0 end
        end
        cursor = data.nextPageCursor or ""
        if cursor == "" then break end
    end
    return list
end

        elseif mode == "buy" then head = head .. "ของครบแล้ว · กำลังอัพ " .. tname
        elseif mode == "save" then head = head .. "รอ " .. col("FFA53A", "เงิน") .. " สำหรับ " .. tname .. " · พร้อมอีก " .. mmss(moneyEta)
        elseif mode == "power" then
            if crewOk or crewWhy == "รอบถัดไป" then head = head .. "รอ " .. col("FFA53A", "พลังทีม") .. " สำหรับ " .. tname .. " · กำลังอัพตัว/อีเตอร์/Core/Mutation เพิ่มพลัง"
            else head = head .. "รอ " .. col("FFA53A", "พลังทีม") .. " สำหรับ " .. tname .. " · อัพตัวไม่ได้: " .. col(crewCol, crewWhy) end
        elseif mode == "zone" then
            if not target.cost then head = head .. "รอ " .. tname .. " (อ่านราคาไม่ได้)"
            elseif money >= target.cost then head = head .. "ของครบแล้ว · กำลัง" .. tname
            else head = head .. "รอ " .. col("FFA53A", "เงิน") .. " สำหรับ " .. tname .. " · พร้อมอีก " .. mmss(moneyEta) end
        else head = head .. (nk and "ไม่มีวิจัยให้ซื้อตอนนี้ · ใช้เงินอัพตัวได้" or col("6DFF7A", "ครบทุกโซนแล้ว")) end
        -- [แก้ v19] โหนดลำดับแรกติดพลังแต่อัพตัวเพิ่มพลังไม่ได้ → บอกเหตุผล (ระบบเลยซื้ออันถูกสุดไปก่อน)
        if (mode == "buy" or mode == "save") and br.first and br.first.node then
            head = head .. " · " .. col("FF7A7A", esc(br.first.node.name) .. " ติดพลัง: " .. crewWhyTh(br.first.why))
        end
        headL.Text = head
        line2.Text = "อัพแล้ว " .. bought .. " · รอของ " .. math.floor(waited) .. " วิ · คิว " .. queue .. " · ล็อก " .. locked ..
            " · อัพตัว " .. (crewOk and col("6DFF7A", "ทำงาน") or col(crewCol, (crewWhy == "ปิดอยู่") and "ปิด" or ("รอ: " .. crewWhy)))
        local oreKinds = 0
        for _, v1 in pairs(r.Ores or {}) do if type(v1) == "number" and v1 > 0 then oreKinds = oreKinds + 1 end end
        line3.Text = "สกิล Lv." .. lvSum .. "/" .. lvMax .. " (" .. tostring(tzone) .. ") · ไอเท็ม " .. oreKinds .. " ชนิด · $" .. big(money)
        -- [แก้ v20] สถานะโพชั่น
        if b.Flags.auto_potion then
            local pOn, pAll, pBan = p.__potStatus(r)
            if pAll > 0 then
                line3.Text = line3.Text .. " · โพชั่น " .. col(pOn == pAll and "6DFF7A" or "FFA53A", pOn .. "/" .. pAll) ..
                    (pBan and (" " .. col("FF7A7A", "เกมไม่รับ " .. esc(pBan))) or "")
            end
        end

        local need = {}
        if target then
            if target.cost then
                local done = money >= target.cost
                need[#need + 1] = "1. เงิน $" .. big(math.min(money, target.cost)) .. "/$" .. big(target.cost) .. " · " ..
                    (done and col("6DFF7A", "✓ ครบ") or ((mode == "power") and ("อีก " .. mmss(moneyEta)) or col("FFA53A", "อีก " .. mmss(moneyEta))))
            else
                need[#need + 1] = "1. เงิน · " .. col("CDD5E0", "อ่านราคาไม่ได้")
            end
            if target.req then
                local have = tonumber(br.power) or 0
                local okp = have >= target.req
                need[#need + 1] = "2. พลังทีม " .. big(math.min(have, target.req)) .. "/" .. big(target.req) .. " · " ..
                    (okp and col("6DFF7A", "✓ ครบ") or col("FFA53A", "ยังไม่ถึง"))
            end
        else
            need[1] = col("6DFF7A", "ไม่มี")
        end
        needL.Text = table.concat(need, "\n")

        -- ไอเท็ม (แร่) + อัตรา/วิ
        local list = {}
        for k1, v1 in pairs(r.Ores or {}) do
            if type(v1) == "number" then
                hist.ores[k1] = hist.ores[k1] or {}
                pushHist(hist.ores[k1], v1)
                if v1 > 0 then list[#list + 1] = { k1, v1, incRate(hist.ores[k1]) } end
            end
        end
        table.sort(list, function(a1, a2) return a1[2] > a2[2] end)
        local lines = {}
        for i1 = 1, math.min(8, #list) do lines[#lines + 1] = i1 .. ". " .. esc(list[i1][1]) .. " " .. big(list[i1][2]) .. " · " .. rateTxt(list[i1][3]) end
        itemsL.Text = (#lines > 0) and table.concat(lines, "\n") or col("CDD5E0", "ยังไม่มีแร่")

        moneyL.Text = "$" .. big(money)
        local etaTxt = "--:--"
        if mode == "save" or (mode == "zone" and target.cost) then etaTxt = mmss(moneyEta)
        elseif mode == "buy" then etaTxt = "00:00" end
        etaL.Text = 'next upgrade in <font size="32">' .. etaTxt .. "</font>"
    end
    b.__threads[#b.__threads + 1] = task.spawn(function()
        while not b.Unloaded and sg.Parent do
            local ok, err = pcall(refresh)
            if not ok then headL.Text = col("FFA53A", "HUD error: " .. esc(err)) end
            task.wait(1)
        end
    end)

    -- ป้ายบนหิน (ESP) — เปิดใน CONFIG.node_esp
    if b.Flags.node_esp then
        local tags = {}
        b.__threads[#b.__threads + 1] = task.spawn(function()
            while not b.Unloaded do
                pcall(function()
                    local nodes0 = T()
                    local seen = {}
                    local cv = b.Flags.esp_colour
                    local colr = (type(cv) == "table" and Color3.new(cv[1] or 0, cv[2] or 0, cv[3] or 0)) or Color3.fromRGB(34, 211, 238)
                    for _, nd in ipairs(nodes0) do
                        seen[nd] = true
                        local part = (nd:IsA("Model") and (nd.PrimaryPart or nd:FindFirstChildWhichIsA("BasePart"))) or nd
                        if part then
                            local bb = tags[nd]
                            if not bb or not bb.Parent then
                                bb = Instance.new("BillboardGui")
                                bb.Name = "NodeTag" bb.AlwaysOnTop = true bb.Size = UDim2.fromOffset(190, 46)
                                bb.StudsOffsetWorldSpace = Vector3.new(0, 4, 0) bb.Adornee = part bb.Parent = sg
                                local t1 = Instance.new("TextLabel") t1.Name = "Title" t1.BackgroundTransparency = 1 t1.Size = UDim2.new(1, 0, 0, 16)
                                t1.Font = Enum.Font.GothamBold t1.TextSize = 13 t1.TextStrokeTransparency = .4 t1.Parent = bb
                                local t2 = Instance.new("TextLabel") t2.Name = "Sub" t2.BackgroundTransparency = 1 t2.Position = UDim2.fromOffset(0, 16)
                                t2.Size = UDim2.new(1, 0, 0, 14) t2.Font = Enum.Font.Gotham t2.TextSize = 12 t2.TextColor3 = Color3.fromRGB(220, 220, 230) t2.TextStrokeTransparency = .5 t2.Parent = bb
                                local bg = Instance.new("Frame") bg.Name = "BarBg" bg.AnchorPoint = Vector2.new(.5, 0) bg.Position = UDim2.new(.5, 0, 0, 33)
                                bg.Size = UDim2.fromOffset(120, 4) bg.BackgroundColor3 = Color3.fromRGB(20, 22, 28) bg.BorderSizePixel = 0 bg.Parent = bb
                                local br = Instance.new("Frame") br.Name = "Bar" br.Size = UDim2.fromScale(1, 1) br.BorderSizePixel = 0 br.Parent = bg
                                tags[nd] = bb
                            end
                            local hp = nd:GetAttribute("NodeHP") or 0
                            local mx = math.max(nd:GetAttribute("NodeMaxHP") or 1, 1)
                            local fr = math.clamp(hp / mx, 0, 1)
                            bb.Title.Text = tostring(nd:GetAttribute("NodeTitle") or nd.Name)
                            bb.Title.TextColor3 = colr
                            bb.Sub.Visible = b.Flags.esp_ore ~= false
                            bb.Sub.Text = ("%s  ·  %s  ·  %d%%"):format(tostring(nd:GetAttribute("NodeOre") or "?"), tostring(nd:GetAttribute("NodeTier") or "?"), math.floor(fr * 100 + .5))
                            bb.BarBg.Visible = b.Flags.esp_hp ~= false
                            bb.BarBg.Bar.Size = UDim2.fromScale(fr, 1)
                            bb.BarBg.Bar.BackgroundColor3 = colr
                        end
                    end
                    for nd, bb in pairs(tags) do if not seen[nd] then pcall(function() bb:Destroy() end) tags[nd] = nil end end
                end)
                task.wait(.5)
            end
        end)
    end
end
